# ---------------------------------------------------------------------------
# Phase 2 — classify_url() unit tests (INP-02)
# ---------------------------------------------------------------------------

def test_classify_url_spotify_track():
    assert classify_url("https://open.spotify.com/track/4cOdK2wGLETKBW3PvgPWqT") == "Spotify"

def test_classify_url_spotify_all_types():
    for path in ["track", "album", "playlist", "artist"]:
        assert classify_url(f"https://open.spotify.com/{path}/abc123") == "Spotify"

def test_classify_url_youtube_watch():
    assert classify_url("https://www.youtube.com/watch?v=dQw4w9WgXcQ") == "YouTube"

def test_classify_url_youtube_short():
    assert classify_url("https://youtu.be/dQw4w9WgXcQ") == "YouTube"

def test_classify_url_youtube_with_list_param():
    assert classify_url("https://youtube.com/watch?v=abc&list=xyz") == "YouTube"

def test_classify_url_invalid_returns_none():
    assert classify_url("https://bad-url.com/whatever") is None
    assert classify_url("not a url") is None
    assert classify_url("") is None


# ---------------------------------------------------------------------------
# Phase 2 — paste → row integration tests (INP-01, INP-02, INP-03)
# ---------------------------------------------------------------------------

def test_paste_populates_one_row_per_url(app_root):
    """INP-01: three non-blank lines → three rows."""
    app_root.clipboard_clear()
    app_root.clipboard_append(
        "https://open.spotify.com/track/abc\n"
        "https://youtu.be/xyz\n"
        "bad-url"
    )
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    assert len(app_root.table.tree.get_children()) == 3


def test_paste_skips_blank_lines(app_root):
    """INP-01: blank lines between URLs are not added as rows."""
    app_root.clipboard_clear()
    app_root.clipboard_append(
        "https://open.spotify.com/track/abc\n"
        "\n"
        "https://youtu.be/xyz\n"
        "\n"
    )
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    assert len(app_root.table.tree.get_children()) == 2


def test_type_badges_after_paste(app_root):
    """INP-02: Spotify and YouTube badges appear immediately after paste."""
    app_root.clipboard_clear()
    app_root.clipboard_append(
        "https://open.spotify.com/track/abc\n"
        "https://youtu.be/xyz"
    )
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    iids = app_root.table.tree.get_children()
    assert app_root.table.tree.set(iids[0], "type") == "Spotify"
    assert app_root.table.tree.set(iids[1], "type") == "YouTube"


def test_invalid_url_shows_error_row(app_root):
    """INP-03: unrecognized URL shows 'Invalid URL' and 'Skipped — bad URL'."""
    app_root.clipboard_clear()
    app_root.clipboard_append("bad-url")
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    iids = app_root.table.tree.get_children()
    assert len(iids) == 1
    assert app_root.table.tree.set(iids[0], "type") == "Invalid URL"
    assert app_root.table.tree.set(iids[0], "status") == "Skipped — bad URL"


def test_invalid_row_does_not_block_valid_rows(app_root):
    """INP-03: mixed batch — invalid row is red, valid rows remain Queued."""
    app_root.clipboard_clear()
    app_root.clipboard_append(
        "https://open.spotify.com/track/abc\n"
        "bad-url\n"
        "https://youtu.be/xyz"
    )
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    iids = app_root.table.tree.get_children()
    assert len(iids) == 3
    # valid rows are Queued
    assert app_root.table.tree.set(iids[0], "status") == "Queued"
    assert app_root.table.tree.set(iids[2], "status") == "Queued"
    # invalid row is flagged
    assert app_root.table.tree.set(iids[1], "type") == "Invalid URL"
    assert app_root.table.tree.set(iids[1], "status") == "Skipped — bad URL"


def test_paste_box_cleared_after_paste(app_root):
    """INP-01: widget is empty (or placeholder-only) after paste fires."""
    app_root.clipboard_clear()
    app_root.clipboard_append("https://open.spotify.com/track/abc")
    app_root.update()
    app_root._paste_box.event_generate("<<Paste>>")
    app_root.update()
    content = app_root._paste_box.get("1.0", "end-1c").strip()
    # either empty or showing placeholder text — NOT the URL
    assert "open.spotify.com" not in content


def test_tag_colors_configured(app_root):
    """INP-02: spotify, youtube, error tags are registered with correct foreground colors."""
    tree = app_root.table.tree
    assert tree.tag_configure("spotify")["foreground"] == "#1DB954"
    assert tree.tag_configure("youtube")["foreground"] == "#EF4444"
    assert tree.tag_configure("error")["foreground"] == "#EF4444"
