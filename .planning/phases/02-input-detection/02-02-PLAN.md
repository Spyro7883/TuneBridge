```python
def classify_url(url: str) -> str | None:
    """Return 'Spotify', 'YouTube', or None for unrecognized/blank URLs."""
    if not url:
        return None
    if _SPOTIFY_RE.search(url):
        return "Spotify"
    if _YOUTUBE_RE.search(url):
        return "YouTube"
    return None
```

```python
def add_row(self, url: str, title: str = "", url_type: str = "") -> str:
    """Add a row and return its iid."""
    if title:
        display_title = title
    else:
        display_title = (url[:60] + "...") if len(url) > 60 else url
    if url_type == "Spotify":
        type_tag = "spotify"
    elif url_type == "YouTube":
        type_tag = "youtube"
    elif url_type == "Invalid URL":
        type_tag = "error"
    else:
        type_tag = ""
    tags = ("queued", type_tag) if type_tag else ("queued",)
    iid = self.tree.insert(
        "", "end",
        values=(display_title, url_type, "Queued"),
        tags=tags,
    )
    self._rows[iid] = url
    if type_tag:
        self._type_tags[iid] = type_tag
    return iid
```

```python
def update_row_status(self, row_id: str, status: str) -> None:
    """Update row status. Must be called from main thread (via after())."""
    if not self.tree.exists(row_id):
        return
    status_tag = self._STATUS_TAG.get(status, "active")
    type_tag = self._type_tags.get(row_id, "")
    tags = (status_tag, type_tag) if type_tag else (status_tag,)
    self.tree.set(row_id, "status", status)
    self.tree.item(row_id, tags=tags)
```

```python
def _on_focus_in(self, event) -> None:
    if self._placeholder_active:
        self._paste_box.delete("1.0", "end")
        self._paste_box.configure(fg="#FFFFFF")
        self._placeholder_active = False

def _on_focus_out(self, event) -> None:
    if not self._paste_box.get("1.0", "end-1c").strip():
        self._paste_box.insert("1.0", self._placeholder_text)
        self._paste_box.configure(fg="#555555")
        self._placeholder_active = True

def _on_paste(self, event) -> str:
    if self._placeholder_active:
        self._paste_box.configure(fg="#FFFFFF")
        self._placeholder_active = False
    try:
        raw = self.clipboard_get()
    except tk.TclError:
        return "break"
    self._process_urls(raw)
    return "break"

def _process_urls(self, raw: str) -> None:
    """Split raw text into URL candidates, classify, and add rows."""
    lines = [line.strip() for line in raw.splitlines()]
    candidates = [line for line in lines if line]
    if not candidates:
        return

    valid_count = 0
    for url in candidates:
        url_type = classify_url(url)
        if url_type is not None:
            self.table.add_row(url=url, url_type=url_type)
            valid_count += 1
        else:
            iid = self.table.add_row(url=url, url_type="Invalid URL")
            self.table.update_row_status(iid, "Skipped — bad URL")

    if valid_count > 0:
        self._status_var.set(
            f"{valid_count} URL(s) added — paste more or start processing"
        )
    else:
        self._status_var.set("No valid URLs found — check your links")

    # Restore placeholder (widget is empty because return 'break' suppressed insertion)
    self._paste_box.insert("1.0", self._placeholder_text)
    self._paste_box.configure(fg="#555555")
    self._placeholder_active = True
```

**Important:** `classify_url` is module-level — call it as `classify_url(url)` not `self.classify_url(url)`.

**Important:** Do NOT call `self._paste_box.delete(...)` before restoring placeholder — the widget is already empty because `return 'break'` in `_on_paste` suppressed default text insertion.
